cmake_minimum_required(VERSION 3.22)
project(image_check LANGUAGES CXX)
find_package(OpenCV REQUIRED CONFIG COMPONENTS world)
find_package(assimp REQUIRED CONFIG)
add_library(image_check SHARED check.cpp)
target_compile_features(image_check PRIVATE cxx_std_17)
target_include_directories(image_check PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(image_check PRIVATE ${OpenCV_LIBS} assimp::assimp)
target_link_options(image_check PRIVATE -Wl,--no-undefined)
